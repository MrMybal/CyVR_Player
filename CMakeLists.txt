cmake_minimum_required(VERSION 3.22)
project(CyVR_Player VERSION 0.4.9 LANGUAGES CXX)

option(CYVR_PLAYER_BUILD_TESTS "Build portable media core tests" ON)
option(CYVR_PLAYER_BUILD_SHADERS "Compile Vulkan 1.1 post-processing kernels" ON)
option(CYVR_PLAYER_WITH_FFMPEG "Build file/stream demuxing and explicit software decoding" OFF)
option(CYVR_PLAYER_BUILD_APP "Build the CyVR_UI application" OFF)
option(CYVR_ENABLE_IMMERSIVE "Offer optional immersive video playback" ON)

add_library(cyvr_player_core STATIC
    src/media.cpp src/video_layout.cpp src/source.cpp src/file_browser.cpp src/postprocess_cas.cpp src/postprocess_fsr.cpp src/audio_queue.cpp)
target_compile_features(cyvr_player_core PUBLIC cxx_std_17)
target_include_directories(cyvr_player_core PUBLIC include)
target_include_directories(cyvr_player_core SYSTEM PRIVATE third_party)
set_target_properties(cyvr_player_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(cyvr_player_core PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(cyvr_player_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CYVR_PLAYER_BUILD_APP)
    if(NOT CYVR_PLAYER_WITH_FFMPEG)
        message(FATAL_ERROR "The application needs CYVR_PLAYER_WITH_FFMPEG=ON")
    endif()
    enable_language(C)
    set(CYVR_UI_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../CyVR_UI" CACHE PATH "CyVR_UI source directory")
    foreach(source core ui spatial gpu)
        if(NOT EXISTS "${CYVR_UI_ROOT}/src/${source}.cpp")
            message(FATAL_ERROR "Missing CyVR_UI component: ${source}")
        endif()
    endforeach()
    if(ANDROID)
        file(GLOB shader_tool_dirs "${ANDROID_NDK}/shader-tools/*")
    endif()
    find_program(CYVR_APP_GLSLC NAMES glslc HINTS ${shader_tool_dirs} REQUIRED)
    set(CYVR_GLSLC "${CYVR_APP_GLSLC}" CACHE FILEPATH "CyVR_UI shader compiler" FORCE)
    set(CYVR_BUILD_REFERENCE OFF CACHE BOOL "Build CyVR_UI reference app" FORCE)
    set(CYVR_INCLUDE_SCENES ${CYVR_ENABLE_IMMERSIVE} CACHE BOOL "Bundle the MIT foundation's CC0 backgrounds" FORCE)
    set(CYVR_BUILD_TESTS "${CYVR_PLAYER_BUILD_TESTS}" CACHE BOOL "Build CyVR_UI tests" FORCE)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
    add_subdirectory("${CYVR_UI_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/cyvr-ui" EXCLUDE_FROM_ALL)
    add_library(cyvr_player_ui ALIAS cyvr_gpu)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/assets")
    set(app_assets)
    foreach(shader video.vert video.frag video_filter.frag ambilight.frag)
        set(output "${CMAKE_CURRENT_BINARY_DIR}/assets/${shader}.spv")
        add_custom_command(OUTPUT "${output}" COMMAND "${CYVR_APP_GLSLC}" --target-env=vulkan1.1 -O
            "-I${CMAKE_CURRENT_SOURCE_DIR}/third_party"
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" -o "${output}" DEPENDS "shaders/${shader}" shaders/image_controls.glsl shaders/gsr1.glsl
            third_party/cas/ffx_cas.h third_party/fsr1/ffx_fsr1.h VERBATIM)
        list(APPEND app_assets "${output}")
    endforeach()
    add_custom_target(cyvr_player_app_assets ALL
        COMMAND "${CMAKE_COMMAND}" -E copy_directory
            "${CMAKE_CURRENT_BINARY_DIR}/cyvr-ui/assets" "${CMAKE_CURRENT_BINARY_DIR}/assets"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${CYVR_UI_ROOT}/assets/fonts/NotoSans-Regular.ttf" "${CMAKE_CURRENT_BINARY_DIR}/assets/subtitle-font.ttf"
        DEPENDS ${app_assets})
    add_dependencies(cyvr_player_app_assets cyvr_assets)
    add_library(cyvr_player_application STATIC src/app/application.cpp src/app/previews.cpp src/app/views.cpp src/app/tracks.cpp src/app/session.cpp src/app/state.cpp src/app/locale.cpp)
    add_dependencies(cyvr_player_application cyvr_player_app_assets)
    target_link_libraries(cyvr_player_application PUBLIC cyvr_player_ui cyvr_player_media)
    include(cmake/Subtitles.cmake)
    include(cmake/Splats.cmake)
    target_link_libraries(cyvr_player_application PUBLIC cyvr_subtitle_render)
    target_link_libraries(cyvr_player_application PUBLIC cyvr_splat_backend)
    add_executable(cyvr_disc_session_probe tools/disc_session_probe.cpp)
    target_link_libraries(cyvr_disc_session_probe PRIVATE cyvr_player_application)
    target_sources(cyvr_player_application PRIVATE src/app/image_processor.cpp src/app/ambilight.cpp src/app/gpu_timer.cpp src/splats.cpp src/splats_io.cpp src/app/splat_renderer.cpp)
    set_target_properties(cyvr_player_application PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(CYVR_PLAYER_BUILD_TESTS AND NOT ANDROID)
        enable_testing()
        add_executable(cyvr_application_tests tests/application_tests.cpp)
        target_link_libraries(cyvr_application_tests PRIVATE cyvr_player_application)
        if(TARGET cyvr_tests)
            add_dependencies(cyvr_application_tests cyvr_tests)
        endif()
        add_test(NAME cyvr_player_application COMMAND cyvr_application_tests)
        add_executable(cyvr_layout_tests tests/video_layout_tests.cpp)
        target_link_libraries(cyvr_layout_tests PRIVATE cyvr_player_core)
        add_test(NAME cyvr_player_layout COMMAND cyvr_layout_tests)
        add_executable(cyvr_placement_tests tests/video_placement_tests.cpp)
        target_link_libraries(cyvr_placement_tests PRIVATE cyvr_player_application)
        add_test(NAME cyvr_player_placement COMMAND cyvr_placement_tests)
    endif()
    if(ANDROID)
        target_sources(cyvr_player_application PRIVATE src/android/software_video.cpp src/android/audio_playback.cpp)
        target_link_libraries(cyvr_player_application PUBLIC aaudio)
        target_link_libraries(cyvr_player_application PUBLIC cyvr_player_android)
        add_library(cyvr_player SHARED src/app/android_main.cpp src/app/android_bridge.cpp src/app/video_renderer.cpp src/app/android_renderer.cpp)
        if(CYVR_ENABLE_IMMERSIVE)
            target_include_directories(cyvr_player PRIVATE "${CYVR_UI_ROOT}/third_party/openxr/include")
            target_compile_definitions(cyvr_player PRIVATE XR_USE_GRAPHICS_API_VULKAN)
        endif()
        target_compile_definitions(cyvr_player PRIVATE VK_USE_PLATFORM_ANDROID_KHR)
        target_link_libraries(cyvr_player PRIVATE cyvr_player_application cyvr_android)
        target_link_options(cyvr_player PRIVATE -uANativeActivity_onCreate -Wl,--no-undefined -Wl,-z,max-page-size=16384)
        add_executable(cyvr_video_gpu_probe tools/video_gpu_probe.cpp src/app/video_renderer.cpp)
        add_executable(cyvr_video_cache_probe tools/video_cache_probe.cpp src/app/video_renderer.cpp)
        target_compile_definitions(cyvr_video_cache_probe PRIVATE VK_USE_PLATFORM_ANDROID_KHR)
        target_link_libraries(cyvr_video_cache_probe PRIVATE cyvr_player_application android log dl)
        add_executable(cyvr_image_probe tools/image_probe.cpp)
        add_executable(cyvr_splat_probe tools/splat_probe.cpp)
        target_link_libraries(cyvr_splat_probe PRIVATE cyvr_player_application android log dl)
        add_executable(cyvr_container_probe tools/container_probe.cpp)
        target_link_libraries(cyvr_container_probe PRIVATE cyvr_player_application android log dl)
        target_link_libraries(cyvr_image_probe PRIVATE cyvr_player_application android log dl)
        target_compile_definitions(cyvr_video_gpu_probe PRIVATE VK_USE_PLATFORM_ANDROID_KHR)
        target_link_libraries(cyvr_video_gpu_probe PRIVATE cyvr_player_application android log dl)
    elseif(WIN32)
        add_executable(CyVR_Player src/app/desktop_main.cpp)
        target_link_libraries(CyVR_Player PRIVATE cyvr_player_application user32 gdi32)
        if(CYVR_PLAYER_BUILD_TESTS)
            add_executable(cyvr_cinema_probe tools/cinema_probe.cpp)
            target_link_libraries(cyvr_cinema_probe PRIVATE cyvr_player_application)
            add_executable(cyvr_image_probe tools/image_probe.cpp)
            add_executable(cyvr_container_probe tools/container_probe.cpp)
            target_link_libraries(cyvr_container_probe PRIVATE cyvr_player_application)
            add_executable(cyvr_splat_probe tools/splat_probe.cpp)
            target_link_libraries(cyvr_splat_probe PRIVATE cyvr_player_application)
            add_test(NAME cyvr_player_gaussian COMMAND cyvr_splat_probe "${CMAKE_CURRENT_BINARY_DIR}/assets")
            target_link_libraries(cyvr_image_probe PRIVATE cyvr_player_application)
            add_test(NAME cyvr_player_gpu_image COMMAND cyvr_image_probe "${CMAKE_CURRENT_BINARY_DIR}/assets")
        endif()
    endif()
endif()

if(CYVR_PLAYER_BUILD_SHADERS)
    if(ANDROID)
        file(GLOB shader_tool_dirs "${ANDROID_NDK}/shader-tools/*")
    endif()
    find_program(CYVR_PLAYER_GLSLC NAMES glslc HINTS ${shader_tool_dirs} REQUIRED)
    find_program(CYVR_PLAYER_SPIRV_VAL NAMES spirv-val HINTS ${shader_tool_dirs})
    set(shader_outputs)
    foreach(shader cas fsr_easu fsr_rcas)
        set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.comp.spv")
        add_custom_command(OUTPUT "${output}"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
            COMMAND "${CYVR_PLAYER_GLSLC}" --target-env=vulkan1.1 -O
                "-I${CMAKE_CURRENT_SOURCE_DIR}/third_party"
                "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.comp" -o "${output}"
            DEPENDS "shaders/${shader}.comp"
                third_party/cas/ffx_a.h third_party/cas/ffx_cas.h
                third_party/fsr1/ffx_a.h third_party/fsr1/ffx_fsr1.h
            VERBATIM)
        list(APPEND shader_outputs "${output}")
    endforeach()
    add_custom_target(cyvr_player_shaders ALL DEPENDS ${shader_outputs})
    if(CYVR_PLAYER_SPIRV_VAL)
        foreach(output IN LISTS shader_outputs)
            add_custom_command(TARGET cyvr_player_shaders POST_BUILD
                COMMAND "${CYVR_PLAYER_SPIRV_VAL}" --target-env vulkan1.1 "${output}"
                VERBATIM)
        endforeach()
    endif()
endif()
